Direct Empirical Scaling Fit With GSE164378 3’ CITE-seq

This tutorial runs the direct empirical scaling workflow on the same GSE164378 3’ CITE-seq RNA-ADT example used by the Gaussian spike tutorials. Here we do not fit the spike model. We sample a grid of cell numbers and UMI depths, compute fixed-rank empirical RNA-ADT mutual information, and fit the empirical scaling law directly.

library(scScale)
## Loading required package: Matrix
data(gse164378_3p_citeseq_hvg)

rna_counts <- gse164378_3p_citeseq_hvg$rna_counts
adt_counts <- gse164378_3p_citeseq_hvg$adt_counts

available_cores <- parallel::detectCores(logical = TRUE)
n_workers <- if (is.na(available_cores)) 2L else max(1L, min(4L, available_cores))

Fit Empirical Scaling Law

fit <- scscale_empirical_scaling_fit(
  rna_counts,
  adt_counts,
  r = 12,
  joint_mode = "hybrid",
  seed = 164378,
  count_transform = "log1p_cpm",
  n_workers = n_workers
)

Inspect The Fit

empirical_mi <- scscale_empirical_values(fit)
empirical_mi_summary <- scscale_empirical_values(fit, summarize = TRUE)
parameters <- scscale_empirical_parameters(fit)

Predict Scaling Laws

cell_curve <- scscale_predict_empirical_cell(
  fit,
  cell_sampling_rate = seq(0.125, 1, length.out = 200)
)

umi_curve <- scscale_predict_empirical_umi(
  fit,
  sampling_rate = seq(0.10, 1, length.out = 200)
)

joint_curve <- scscale_predict_empirical_joint(
  fit,
  cell_sampling_rate = seq(0.125, 1, length.out = 200),
  sampling_rate = c(0.20, 0.50, 1.00)
)
